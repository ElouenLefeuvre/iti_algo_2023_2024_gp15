     \begin{tadOperations}{obtenirCodeBinaire}
        \tadOperation{obtenirCodeBinaire}{\tadDeuxParams{TableDeCodage}{Caractere}}{\tadParam{CodeBinaire}}

        \end{tadPreconditions}
