\section{TAD Code Binaire}
\begin{tad}
     \tadNom{Code Binaire}
        
     \tadDependances{TableDeCodage, Caractere, Booléen }
     \begin{tadOperations}{obtenirCodeBinaire}
        \tadOperation{codeBinaire}{}{\tadParams{CodeBinare}}
        \tadOperation{obtenirCodeBinaire}{\tadDeuxParams{TableDeCodage}{Caractere}}{\tadParams{CodeBinaire}}
        \tadOperationAvecPreconditions{estBinaire}{\tadParams{CodeBinaire}}{\tadParams{Booléen}}
     \end{tadOperations}
      \begin{tadSemantiques}{}
            \tadSemantique{obtenirCodeBinaire}{obtient grâce à la table de codage le code binaire d'un caractère.}
    \tadSemantique{estBinaire}{Permet de vérifier que le code est bien binaire.}
         
      \end{tadSemantiques}
    \begin{tadAxiomes}
        \tadAxiome{estBinaire(obtenirCodeBinaire(TableDeCodage))}
    \end{tadAxiomes}
    \begin{tadPreconditions}{estBinaire(code)}
        \tadPrecondition{estBinaire(code)}{non estVide(code)}
    \end{tadPreconditions}

\end{tad}   


\section{TAD Octet}
\begin{tad}
\tadNom{Octet}
\tadDependances{Chaine de caracteres, Code Binaire, Booleen}
\begin{tadOperations}{estOctet}
    \tadOperation{estOctet}{\tadParams{Octet}}{\tadParams{Booleen}}
    \tadOperation{encoder}{\tadDeuxParams{Octet}{Code Binaire}}{\tadParams{ Chaine de caracteres}}
    \tadOperation{decoder}{\tadDeuxParams{Chaine de caracteres}{Code Binaire}}{\tadParams{Octet}}
    \tadOperation{obtenirOctet}{\tadParams{Chaine de caracteres}}{\tadParams{Octet}}
\end{tadOperations}
\begin{tadSemantiques}{}
    \tadSemantique{estOctet}{vérifie si l'élément est bien un octet.}
    \tadSemantique{encoder}{donne la chaîne de caractères associée à un Octet et un Code Binaire.}
    \tadSemantique{decoder}{donne l'octet associé à la Chaîne de caractères et un Code Binaire.}
    \tadSemantique{obtenirOctet}{donne l'octet associé à une Chaîne de caractères.}
         
\end{tadSemantiques}
\begin{tadAxiomes}
        \tadAxiome{estOctet(obtenirOctet(CDC))}
        \tadAxiome{estOctet(decoder(CDC,CodeBinaire))}
        \tadAxiome{decoder(encoder(Octet,CodeBinaire)) = Octet}
\end{tadAxiomes}
\end{tad}
