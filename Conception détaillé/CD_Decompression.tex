\begin{algorithme}
    \fonction{decompresser}{fic : FichierHuffman}{FichierTexte}
    {}
    {stats\_fic : Ensemble de Statistiques, file : FileDePriorite, temp : ArbreHuffman, digit,car : Caractere}
{
    \affecter{stats\_fic}{obtenirStatistiques(fic)}
    \affecter{file}{obtenirFileDePriorite(stats\_fic)}
    \affecter{temp}{obtenirElement(file)}\newline
    ouvrir(fic)
    \tantque{non(eof(fic))}{
        ouvrir(fic)\newline
        \tantque{non(estUneFeuille(temp))}{
            \affecter{digit}{\lire{fic,car}}
            \sialorssinon{digit==0}
                {\affecter{temp}{obtenirFilsGauche(arbre)}}
                {\affecter{temp}{obtenirFilsDroit(arbre)}}}
        fermer(fic)
        ouvrir(fictexte)
        \ecrire{fictexte, obtenirCaractere(temp)}  \newline
        fermer(fictexte)}
}

\subsubsection{obtenirFileDePriorite}

\begin{algorithme}
    \fonction{ obtenirFileDePriorite}{stats : Ensemble de Statistiques}{FileDePriorite}
    {}
    {ens : Ensemble de Arbre de Huffman, file : FileDePriorite, s : Statistiques, a : ArbreHuffman}
    {
        \affecter{file}{filedePriorite()}
        \pourChaque{s}{stats}
            {ajouter(ens, arbrehuffman(s))}
        \pourChaque{a}{ens}
            {enfiler(file, a)}
        \tantque{longueur$>$1}
            {nouvelleFile(file)}
        \retourner{file}
    }
\end{algorithme}

\subsubsection{obtenirStatistiques}

\begin{algorithme}
    \fonction{obtenirStatistiques}{fic :  FichierHuffman}{Ensemble de Statistiques}
    {}
    {ens : Ensemble de Statistiques}
    {ouvrir(fic) \newline
    \tantque{str!="/"}{
        \lire{fic,car}
        \lire{fic,int}\newline
        ajouter(ens,Statistiques.statistique(car,int))}
    \retourner{ens}
    }
\end{algorithme}
