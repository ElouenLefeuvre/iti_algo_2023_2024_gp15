    \champEnregistrement{filsGauche}{\typePointeur{ArbreHuffman}}
    \champEnregistrement{filsDroit}{\typePointeur{ArbreHuffman}}
\end{enregistrement}
\end{algorithme}


\section{arbreDeHuffman}

\begin{algorithme}

    \fonction{arbreDeHuffman}{stat : Statistiques}{ArbreHuffman}{}
        {arbre : \typePointeur{ArbreHuffman}}
        {
        \sialors{arbre $\neq$ NIL}
        {
        \affecter{\champPointeur{arbre}{statistiques}}{stat}
        \affecter{\champPointeur{arbre}{filsGauche}}{NIL}
        \affecter{\champPointeur{arbre}{filsDroit}}{NIL}
        }
        \retourner{arbre}        
        }
\end{algorithme}

\section{fusionner}

\begin{algorithme}

    \fonction{fusionner}{arbre1 , arbre2 : ArbreHuffman}{ArbreHuffman}{non (estVide(arbre1) et estVide(arbre2))}
        {arbre : ArbreHuffman}
        {
            \affecter{arbre}{arbrehuffman()}
            \affecter{\champ{arbre}\champ{statistiques}{caractere}}{NULL}
            \affecter{\champ{arbre}\champ{statistiques}{ponderation}}{obtenirOccurences(arbre1) + obtenirOccurences(arbre2)}
            \sialorssinon{obtenirOccurences(arbre1)$<$obtenirOccurences(arbre2)}
            {\affecter{\champ{arbre}{filsGauche}}{arbre1}
            \affecter{\champ{arbre}{filsDroit}}{arbre2}}
            {\affecter{\champ{arbre}{filsDroit}}{arbre1}
            \affecter{\champ{arbre}{filsGauche}}{arbre2}}
            \retourner{arbre}
        

        }
\end{algorithme}
