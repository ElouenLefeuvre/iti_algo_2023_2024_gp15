\section{creerFileDePriorite}
\begin{algorithme}
    \fonction{ creerFileDePriorite}{fic :  FichierHuffman}{FileDePriorite}
    {}
    {file : FileDePriorite; ensC : Ensemble de \caractere; ensS : Ensemble de Statistiques}
    {\affecter{ensC}{obtenirCaracteres(fic)}
    \affecter{ensS}{creerStatistique(ensC)}
    \affecter{file}{filedePriorite()}
    \pourChaque{Stats}{ensS}
    {insererArbre(file, creerArbre(stats))}
    
    \tantQue{non(File.estVide(defiler(file)))}  %Ou faire avec longueur >1 de FileDePrio
        {formationNouvelArbrePondere(file)}
    \retourner(file)}
\end{algorithme}

\section{creerStatistique}

\begin{algorithme}
    \fonction{ creerStatistique}{ensC : Ensemble de \caractere}{Ensemble de Statistiques}
    {}
    {ensStats : Ensemble de Statistiques, nbOcc : \naturel}   				
    {\pourChaque{car}{ensC}
        {\affecter{nbOcc}{Statistiques.obtenirOccurence(car)}
        Ensemble.ajouter(ensStats,statistique(car, nbOcc))}
    \retourner{ensStats}}
        
\end{algorithme}
