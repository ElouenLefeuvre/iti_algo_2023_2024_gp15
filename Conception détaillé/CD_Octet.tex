\subsubsection{modifier}

\begin{algorithme}
    \procedure{modifier}{\paramEntreeSorite{o : Octet},\paramEntree{i : \naturel, valeur : Bit}}{}
    {b : Bit}
    {\cas{i}{
        \casclausegenerale{0}{\affecter{o.bit0}{valeur}}
        \casclausegenerale{1}{\affecter{o.bit1}{valeur}}
        \casclausegenerale{2}{\affecter{o.bit2}{valeur}}
        \casclausegenerale{3}{\affecter{o.bit3}{valeur}}

    }
    \retourner{o}}
\end{algorithme}

\subsubsection{creerOctet}

\begin{algorithme}
    \fonction{creerOctet}{}{Octet}
    {}
    {o : Octet
    i : \naturel}
    {\pour{i}{0}{7}{}
        {modifier(o,i,0)}
    \retourner{o}}
\end{algorithme}

\subsubsection{encoder}

\begin{algorithme}
    \procedure{encoder}{\paramEntreeSortie{o : Octet}\paramEntree{code : CodeBinaire, i : \naturel}}
    {non estVide(code), 0 \leq i $<=$ 7}
    {j : \naturel}
    {\pour{j}{i}{i + longueur(code)}{}
    {modifier(o,j,obtenirIemeBit(code,j))}}
\end{algorithme}

\subsubsection{obtenirOctet}


\begin{algorithme}
    \fonction{obtenirOctet}{o : Octet}{\chaine}
    {}
    {ch : \chaine}
    {\pourChaque{bit}{o}
    {\affecter{ch}{ch + bit}}
    \retourner{ch}}
\end{algorithme}
