\subsubsection{bitsToStr}

\begin{algorithme}
    \fonction{bitsToStr}{octet : \typePointeur{Octet}}{\chaine}{}
        {str : \typePointeur{\chaine}, i : \naturel}
        {\instruction{\allouer{str}}
        \pour{i}{1}{8}{}
            {\sialorssinon{\champPointeur{octet}{bits}[0] == BITA0}
                {\affecter{str[i]}{'0'}}
                {\affecter{str[i]}{'1'}}}
                \affecter{str[8]}{'0'}}
\end{algorithme}

\subsubsection{strToBits}

\begin{algorithme}
    \fonction{strToBits}{chaine : \chaine}{Octet}{}
        {octet : Octet, i : \naturel, longueurChaine : \entier}
        {\affecter{octet}{creerOctet()}
        \affecter{longueurChaine}{taille(chaine)}
        \pour{i}{1}{longueurChaine-1}{}
            {\sialorssinon{\champPointeur{chaine[i ]== '0'}}
                {\affecter{\champ{octet}{bits}[i]}{BITA0}
                \instruction{incrementStartPosition()}}
                {\affecter{\champ{octet}{bits}[i]}{BITA1}
                \instruction{incrementStartPosition()}}}}
\end{algorithme}

\subsubsection{encoder}
\fonction{encoder}{octet : Octet,cb : CodeBinaire}{Octet}
    {}
    {bit : Bit}
    {\sialorssinon{mostImportantBitRestant $\le$ 8 et non(currentisFull)}
        {\pour{i}{mostImportantBitRestant}{9}{}
            {\affecter{bit}{iemeBit(bitRestant,i)}
            \instruction{fixerBit(\adresse{octet},startPosition,bit)}
            \affecter{startPosition}{startPosition+1}
            \instruction{fixerBit(\adresse{bitRestant},mostImportantBitRestant,BITA0)}
            \affecter{mostImportantBitRestant}{mostImportantBitRestant+1}}
        }{}
    \pour{i}{0}{\champ{cb}{taille}}{}
        {\affecter{bit}{CB_obtenirIemeBit(cb,i)}
        \sialorssinon{non(currentisFull)}
            {\instruction{fixerBit(\adresse{octet}, startPosition,bit)}
            \affecter{startPosition}{startPosition+1}}
            
            {\instruction{decaler(\adresse{bitRestant})}
            \instruction{fixerBit(\adresse{bitRestant}, 8, bit)}
            \affecter{mostImportantBitRestant}{mostImportantBitRestant-1}}
        \sialorssinon{startPosition$>$8}
        {\affecter{currentisFull}{Vrai}
        }{}
        }
    \sialorssinon{startPosition$>$8}{
        \affecter{currentisFull}{currentisFull+1}}{}
    \retourner{octet}
    }

\subsubsection{inverser}

\begin{algorithme}
