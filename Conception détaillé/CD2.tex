\section{Décompression}

\begin{algorithme}
    \signatureFonction{ decompresser}{fic : FichierHuffman}{FichierTexte}\newLine
    {}
début
    stats_fic = obtenirStatistiques(fic)
    file = obtenirFileDePriorite(stats_fic)
    temp = obtenirElement(file)
    ouvrir(fic)
    tant que non (eof(fic)):
        ouvrir(fic)
        tant que non estUneFeuille(temp)    
            digit = read(fic, car)
            if digit ==0
                temp = obtenirFilsGauche(arbre)
            sinon 
                temp = obtenirFilsDroit(arbre)
            finsi
        fintantque
        fermer(fic)
        ouvrir(fictexte)
        ecrire(fictexte, obtenirCaractere(temp))  
        fermer(fictexte)  
    fintantque
fin



\end{algorithme}

\section{obtenirFileDePriorite}

\begin{algorithme}
    \fonction{ obtenirFileDePriorite}{stats : Ensemble de Statistiques}{FileDePriorite}
    {}
    {ens : Ensemble de Arbre de Huffman}
    {
        file = filedePriorite(NULL)
        ens = ensemble()
        pour chaque s dans stats faire
            ajouter(ens, arbrehuffman(s))
        pour chaque a dans ens faire
            enfiler(file, a)
        tant que longeur(file)>1 faire
            nouvelleFile(file)
        retourner file
    }
\end{algorithme}

\section{obtenirStatistiques}

\begin{algorithme}
    \fonction{obtenirStatistiques}{fic :  FichierHuffman}{Ensemble de Statistiques}
    {}
    {ens : Ensemble de Statistiques}
    {ouvrir(fic)
    \tantque{str!="/"}
        {\lire{fic,car}
        \lire{fic,int}
        ajouter(ens,Statistiques.statistique(car,int))}
    \retourner{ens}
    }
\end{algorithme}

\section{creerArbreHuffman}

\begin{algorithme}
    \fonction{creerArbreHuffman}{stats : Statistiques}{Arbre Huffman}
    {}
    {}
    {\retourner{ArbreHuffman.arbrehuffman(stats)}}

\end{algorithme}

    {}
