\section{Décompression}

\begin{algorithme}
    \signatureFonction{ decompresser}{fic : FichierHuffman}{FichierTexte}\newLine
    {}
début
    stats_fic = obtenirStatistiques(fic)
    pour chaque s dans stats_fic:
        arbre = fusionner(arbre,arbrehuffman())
        fusionner(arbre,s)
    ouvrir(fic)
    tant que non (eof(fic)) et ens!=none:
        ouvrir(fic)
        temp = arbre
        tant que non estUneFeuille(temp)    
            digit = read(fic, car)
            if digit ==0
                temp = obtenirFilsGauche(arbre)
            sinon 
                temp = obtenirFilsDroit(arbre)
            finsi
        fintantque
        fermer(fic)
        ouvrir(fictexte)
        ecrire(fictexte, obtenirCaractere(temp))  
        fermer(fictexte)  
    fintantque
fin



\end{algorithme}

\section{obtenirStatistiques}

\begin{algorithme}
    \fonction{obtenirStatistiques}{fic :  FichierHuffman}{Ensemble de Statistiques}
    {}
    {ens : Ensemble de Statistiques}
    {ouvrir(fic)
    \tantque{str!="/"}
        {\lire{fic,car}
        \lire{fic,int}
        ajouter(ens,Statistiques.statistique(car,int))}
    \retourner{ens}
    }
\end{algorithme}
