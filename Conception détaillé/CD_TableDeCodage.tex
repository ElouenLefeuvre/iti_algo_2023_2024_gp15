\begin{algorithme}
    \fonction{obtenirTableDeCodage}{a : Arbre Huffman}{TableDeCodage}
    {non estVide(a)}
    {res : TableDeCodage}
    
    début
        res.car < obtenirCaractere(a.statistiques)
        res.code < obtenirCodeBinaire(res.car)
        listeSuivante < NIL
        retourner res
    fin

\begin{algorithme}
    \procedure{ajouter}{\paramEntreeSortie tab : TableDeCodage, \paramEntree car : \caractere, code : CodeBinaire}
    {}
    {temp : TableDeCodage}
    début
        temp < tab
        allouer(tab)
        tab.car < car
        tab.code < code
        tab.listeSuivante < temp
    fin
    {}
\end{algorithme}

\begin{algorithme}
    \procedure{supprimer}{\paramEntreeSortie tab : TableDeCodage, \paramEntree car : \caractere, code : CodeBinaire}
    {non(estVide(tab))}
    {temp : TableDeCodage, supprime : \booleen}
    début
        si non(estVide(tab.listeSuivante)) alors
            si temp.car = car alors
                 temp < temp.listeSuivante
                 tab.listeSuivante < temp
            sinon
                 supprimer(temp.listeSuivante,car,code)
                 tab.listeSuivante < temp
            finsi
        finsi
    fin

    {}
\end{algorithme}
