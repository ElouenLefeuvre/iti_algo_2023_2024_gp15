\section{TableDeCodage}

\begin{algorithme}
    \fonction{obtenirTableDeCodage}{a : Arbre Huffman}{TableDeCodage}
    {non estVide(a)}
    {res : TableDeCodage}
    {\affecter{res.car}{obtenirCaractere(a.statistiques)}
    \affecter{res.code}{obtenirCodeBinaire(res.car)} 
    \affecter{res.listeSuivante}{NIL} 
    \retourner{res}}
\end{algorithme}



\begin{algorithme}
    \fonction{obtenirCode}{tab: TableDeCodage, car : \caractere}{CodeBinaire}
    {estPresent(tab,car)}
    {}
    {\sialorssinon{non(estVide(tab))}
        {\sialorssinon{tab.car=car}
            {\retourner{tab.code}}
            {\retourner{obtenirCode(tab.listeSuivante)}}}
        {}}
\end{algorithme}

\begin{algorithme}
    \fonction{obtenirCarac}{tab: TableDeCodage, code : CodeBinaire}{\caractere}
    {}
    {}
    {\sialorssinon{non(estVide(tab))}
        {\sialorssinon{tab.code=code }
            {\retourner{tab.car}}
            {\retourner {obtenirCode(tab.listeSuivante)}}}
        {}}

\end{algorithme}

\begin{algorithme}
    \procedure{ajouter}{\paramEntreeSortie tab : TableDeCodage, \paramEntree car : \caractere, code : CodeBinaire}
    {}
    {temp : TableDeCodage}
    {\affecter{temp}{tab} 
    \affecter{tab.car}{car} 
    \affecter{tab.code}{code}
    \affecter{tab.listeSuivante}{temp}}
\end{algorithme}

\begin{algorithme}
    \procedure{supprimer}{\paramEntreeSortie tab : TableDeCodage, \paramEntree car : \caractere, code : CodeBinaire}
    {non(estVide(tab))}
    {temp : TableDeCodage}
    {\sialorssinon{non(estVide(tab.listeSuivante))}
        {\sialorssinon{temp.car = car}
            {\affecter{temp}{temp.listeSuivante}
            \affecter{tab.listeSuivante}{temp}}
            {supprimer(temp.listeSuivante,car,code)
            \affecter{tab.listeSuivante}{temp}}}
        {}}
\end{algorithme}

\begin{algorithme}
    \fonction{estPresent}{tab : TableDeCodage, car : \caractere}{\booleen}
    {}
    {}
    {\sialorssinon{estVide(tab)}
        {\retourner{FAUX}}{}
    \sialorssinon{tab.car = car}
        {\retourner{VRAI}}
        {\retourner{estPresent(tab.listeSuivante,car)}}}
\end{algorithme}
