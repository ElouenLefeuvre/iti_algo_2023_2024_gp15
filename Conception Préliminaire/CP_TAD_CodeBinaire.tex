\section{TAD Code Binaire}
\begin{algorithme}
\signatureFonction{obtenirCodeBinaire}
	{a : Arbre Huffman, car : \caractere}{CodeBinaire}
	{non(estVide(a))}
\end{algorithme}

\begin{algorithme}
\signatureFonction{estVide}
	{code : CodeBinaire}{\booleen}
	{}
\end{algorithme}

\begin{algorithme}
	\signatureFonction{obtenirIemeBit}
		{code : CodeBinaire, i : \naturelNonNul}{Bit}
		{i > 0 et i <= longueur(code) }
	\end{algorithme}
